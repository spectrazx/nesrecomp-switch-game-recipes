cmake_minimum_required(VERSION 3.20)
project(NESRecompGame C)
set(GAME_ID "" CACHE STRING "Game module directory name")
if(NOT GAME_ID MATCHES "^[a-z0-9_-]+$" OR NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/${GAME_ID}/game.cmake")
    message(FATAL_ERROR "Unknown game module: ${GAME_ID}")
endif()
set(NESRECOMP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.local/src/nesrecomp" CACHE PATH "NESRecomp checkout")
get_filename_component(NESRECOMP_ROOT "${NESRECOMP_ROOT}" ABSOLUTE)
get_filename_component(NESRECOMP_SOURCES_ROOT "${NESRECOMP_ROOT}/.." ABSOLUTE)
set(NESRECOMP_ROM "" CACHE FILEPATH "User-supplied ROM")
set(NESRECOMP_HOST_COMPILER "" CACHE FILEPATH "Runnable code generator")
set(GAME_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/.." CACHE PATH "Packaged NRO directory")
set(GAME_PPU_OVERRIDE "" CACHE STRING "Explicit PPU override for comparisons")
set(default_variant baseline)
if(NINTENDO_SWITCH)
    set(default_variant optimized)
endif()
set(GAME_VARIANT "${default_variant}" CACHE STRING "baseline or optimized")
if(NOT GAME_VARIANT MATCHES "^(baseline|optimized)$")
    message(FATAL_ERROR "Unknown game variant: ${GAME_VARIANT}")
endif()
set(GAME_MODULE_DIR "${CMAKE_CURRENT_LIST_DIR}/${GAME_ID}")
include("${GAME_MODULE_DIR}/game.cmake")
if(NOT GAME_ARTIFACT MATCHES "^[A-Za-z0-9_-]+$")
    message(FATAL_ERROR "Invalid artifact name: ${GAME_ARTIFACT}")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify-rom.py"
    --rom "${NESRECOMP_ROM}" --crc "${GAME_ROM_CRC32}" --mapper "${GAME_ROM_MAPPER}" --title "${GAME_TITLE}"
    RESULT_VARIABLE verified)
if(NOT verified EQUAL 0)
    message(FATAL_ERROR "ROM verification failed for ${GAME_ID}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${NESRECOMP_ROM}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify-rom.py")
if(GAME_VARIANT STREQUAL "optimized")
    set(ppu "${GAME_OPTIMIZED_PPU}")
else()
    set(ppu "${GAME_BASELINE_PPU}")
endif()
if(GAME_PPU_OVERRIDE)
    set(ppu "${GAME_PPU_OVERRIDE}")
endif()
# Selecting a variant also updates a reused CMake cache.
set(NESRECOMP_PPU_IMPL "${ppu}" CACHE STRING "PPU selected by game variant" FORCE)
set(NESRECOMP_SWITCH_TITLE "${GAME_TITLE}" CACHE STRING "NACP title" FORCE)
set(NESRECOMP_SWITCH_GAME_DIR "sdmc:/switch/nesrecomp/${GAME_ID}" CACHE STRING "Runtime SD directory" FORCE)
set(NESRECOMP_RECOMP_UI OFF CACHE STRING "Game modules use the direct runner" FORCE)
include("${NESRECOMP_ROOT}/runner/cyc/project.cmake")
set(host_options "")
if(NOT NINTENDO_SWITCH)
    list(APPEND host_options HEADLESS)
endif()
nesrecomp_add_cycle_game(nes_game ROM "${NESRECOMP_ROM}" GAME_CONFIG "${GAME_MODULE_DIR}/game.toml"
    SEED_FILE "${GAME_CYCLE_SEEDS}" RECOMPILER "${NESRECOMP_HOST_COMPILER}"
    HARDWARE_PROFILE "${GAME_HARDWARE_PROFILE}" OPTIMIZATION_MODULES ${GAME_OPTIMIZATION_MODULES}
    ${host_options})
if(NOT NINTENDO_SWITCH AND GAME_VARIANT STREQUAL "optimized")
    include(CheckIPOSupported)
    check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error LANGUAGES C)
    if(NOT ipo_supported)
        message(FATAL_ERROR "Optimized host variant requires LTO: ${ipo_error}")
    endif()
    set_property(TARGET nes_game PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
if(NINTENDO_SWITCH)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/game-artifact.txt" "${GAME_ARTIFACT}.nro\n")
    add_custom_target(package_game ALL
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${GAME_OUTPUT_DIR}"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_BINARY_DIR}/nes_game.nro"
            "${GAME_OUTPUT_DIR}/${GAME_ARTIFACT}.nro"
        DEPENDS nes_game_nro VERBATIM)
endif()
message(STATUS "Game ${GAME_ID}: ${GAME_VARIANT}, PPU ${ppu}, hardware ${GAME_HARDWARE_PROFILE}")
